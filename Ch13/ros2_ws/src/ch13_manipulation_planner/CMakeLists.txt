cmake_minimum_required(VERSION 3.8)
project(ch13_manipulation_planner)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_srvs REQUIRED)
find_package(pluginlib REQUIRED)
find_package(moveit_ros_move_group REQUIRED)
find_package(moveit_ros_planning REQUIRED)
add_executable(planner_runtime src/planner_runtime.cpp)
target_compile_features(planner_runtime PUBLIC cxx_std_17)
target_link_libraries(planner_runtime moveit_ros_move_group::moveit_move_group_capabilities_base ${CMAKE_DL_LIBS})
ament_target_dependencies(planner_runtime rclcpp std_srvs pluginlib moveit_ros_move_group moveit_ros_planning)
install(TARGETS planner_runtime DESTINATION lib/${PROJECT_NAME})
ament_package()
