cmake_minimum_required(VERSION 3.16)
project(integrated_robot_camera)
find_package(ament_cmake REQUIRED)
find_package(mujoco_vendor REQUIRED)
find_package(Threads REQUIRED)
find_package(mujoco_ros2_control_plugins REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(OpenGL REQUIRED COMPONENTS EGL)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(INTEGRATED_CAMERA_ENABLE_TSAN "Instrument mailbox core and its tests with TSAN" OFF)

add_library(snapshot_mailbox SHARED src/snapshot_mailbox.cpp)
target_include_directories(snapshot_mailbox PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
target_link_libraries(snapshot_mailbox PUBLIC mujoco::mujoco Threads::Threads)
target_compile_options(snapshot_mailbox PRIVATE -Wall -Wextra -Wpedantic)
if(INTEGRATED_CAMERA_ENABLE_TSAN)
  target_compile_options(snapshot_mailbox PUBLIC -fsanitize=thread -g -fno-omit-frame-pointer)
  target_link_options(snapshot_mailbox PUBLIC -fsanitize=thread)
endif()
install(TARGETS snapshot_mailbox EXPORT export_${PROJECT_NAME}
  LIBRARY DESTINATION lib ARCHIVE DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
ament_export_dependencies(mujoco_vendor Threads)

add_library(snapshot_camera_plugin SHARED src/snapshot_camera_plugin.cpp src/egl_render_context.cpp src/camera_trace.cpp)
target_include_directories(snapshot_camera_plugin PRIVATE src)
target_link_libraries(snapshot_camera_plugin PUBLIC
  snapshot_mailbox mujoco_ros2_control_plugins::mujoco_ros2_control_plugins pluginlib::pluginlib rclcpp::rclcpp
  ${sensor_msgs_TARGETS} ${std_srvs_TARGETS} ${std_msgs_TARGETS} OpenGL::EGL OpenGL::OpenGL)
install(TARGETS snapshot_camera_plugin LIBRARY DESTINATION lib)
pluginlib_export_plugin_description_file(mujoco_ros2_control_plugins integrated_robot_camera_plugins.xml)
add_executable(camera_plugin_probe src/camera_plugin_probe.cpp)
target_link_libraries(camera_plugin_probe PRIVATE
  mujoco_ros2_control_plugins::mujoco_ros2_control_plugins pluginlib::pluginlib)
install(TARGETS camera_plugin_probe DESTINATION lib/${PROJECT_NAME})
install(FILES LICENSE REUSE_MANIFEST.md DESTINATION share/${PROJECT_NAME})

# Generated build-time provenance, not a claim that an unbuilt source tree ran.
file(GLOB_RECURSE camera_inputs CONFIGURE_DEPENDS RELATIVE ${CMAKE_CURRENT_SOURCE_DIR}
  include/*.hpp src/*.cpp src/*.hpp test/*.cpp test/*.hpp)
list(APPEND camera_inputs CMakeLists.txt package.xml integrated_robot_camera_plugins.xml LICENSE REUSE_MANIFEST.md)
list(SORT camera_inputs)
set(camera_manifest "{\n")
foreach(input IN LISTS camera_inputs)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${input}")
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${input}" input_hash)
  string(APPEND camera_manifest "  \"${input}\": \"${input_hash}\",\n")
endforeach()
string(REGEX REPLACE ",\n$" "\n" camera_manifest "${camera_manifest}")
string(APPEND camera_manifest "}\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/source_manifest.json" "${camera_manifest}")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/source_manifest.json" DESTINATION share/${PROJECT_NAME})

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_camera_trace test/test_camera_trace.cpp TIMEOUT 30)
  target_include_directories(test_camera_trace PRIVATE src)
  target_link_libraries(test_camera_trace snapshot_camera_plugin ${std_msgs_TARGETS} ${CMAKE_DL_LIBS})
  target_link_options(test_camera_trace PRIVATE -Wl,--export-dynamic)
  ament_add_gtest(test_snapshot_mailbox test/test_snapshot_mailbox.cpp TIMEOUT 30)
  target_include_directories(test_snapshot_mailbox PRIVATE include)
  target_link_libraries(test_snapshot_mailbox snapshot_mailbox)
  ament_add_gtest(test_snapshot_camera_plugin test/test_snapshot_camera_plugin.cpp TIMEOUT 120)
  target_include_directories(test_snapshot_camera_plugin PRIVATE src)
  target_link_libraries(test_snapshot_camera_plugin snapshot_camera_plugin ${CMAKE_DL_LIBS} ${std_msgs_TARGETS})
  target_link_options(test_snapshot_camera_plugin PRIVATE -Wl,--export-dynamic)
  ament_add_gtest(test_egl_context test/test_egl_context.cpp TIMEOUT 180)
  target_include_directories(test_egl_context PRIVATE src)
  target_link_libraries(test_egl_context snapshot_camera_plugin ${CMAKE_DL_LIBS})
  target_link_options(test_egl_context PRIVATE -Wl,--export-dynamic)
  ament_add_gtest(test_plugin_loading test/test_plugin_loading.cpp TIMEOUT 30)
  target_link_libraries(test_plugin_loading mujoco_ros2_control_plugins::mujoco_ros2_control_plugins pluginlib::pluginlib)
endif()
ament_package()
