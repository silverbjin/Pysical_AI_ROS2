cmake_minimum_required(VERSION 3.8)
project(ch13_reuse_wire)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(rosgraph_msgs REQUIRED)
add_executable(ingress src/ingress.cpp)
target_compile_features(ingress PUBLIC cxx_std_17)
ament_target_dependencies(ingress rclcpp std_msgs geometry_msgs nav_msgs sensor_msgs rosgraph_msgs)
install(TARGETS ingress DESTINATION lib/${PROJECT_NAME})
ament_package()
