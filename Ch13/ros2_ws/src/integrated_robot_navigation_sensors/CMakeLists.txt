# Same build-time provenance mechanism as the Ch13 camera package (8e07b05).
file(GLOB_RECURSE sensor_inputs CONFIGURE_DEPENDS RELATIVE ${CMAKE_CURRENT_SOURCE_DIR}
  include/*.hpp src/*.cpp test/*.cpp test/*.hpp)
list(APPEND sensor_inputs CMakeLists.txt package.xml navigation_plugins.xml gazebo_position_plugins.xml LICENSE REUSE_MANIFEST.md)
list(SORT sensor_inputs)
set(sensor_manifest "{\n")
foreach(input IN LISTS sensor_inputs)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${input}")
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${input}" input_hash)
  string(APPEND sensor_manifest "  \"${input}\": \"${input_hash}\",\n")
endforeach()
string(REGEX REPLACE ",\n$" "\n" sensor_manifest "${sensor_manifest}")
string(APPEND sensor_manifest "}\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/source_manifest.json" "${sensor_manifest}")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/source_manifest.json" DESTINATION share/${PROJECT_NAME})
