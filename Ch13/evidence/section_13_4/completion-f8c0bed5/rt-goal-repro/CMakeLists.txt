cmake_minimum_required(VERSION 3.16)
project(ch13_rt_goal_repro)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(control_msgs REQUIRED)
find_package(realtime_tools REQUIRED)
add_executable(rt_goal_original rt_goal_repro.cpp)
ament_target_dependencies(rt_goal_original rclcpp rclcpp_action control_msgs realtime_tools)
target_compile_features(rt_goal_original PRIVATE cxx_std_17)
if(CH13_PATCH_INCLUDE)
  add_executable(rt_goal_patched rt_goal_repro.cpp)
  ament_target_dependencies(rt_goal_patched rclcpp rclcpp_action control_msgs realtime_tools)
  target_include_directories(rt_goal_patched PRIVATE "${CH13_PATCH_INCLUDE}")
  target_compile_definitions(rt_goal_patched PRIVATE CH13_PATCHED)
  target_compile_features(rt_goal_patched PRIVATE cxx_std_17)
endif()
