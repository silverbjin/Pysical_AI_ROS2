cmake_minimum_required(VERSION 3.16)
project(ch13_clock_rate_diagnostic LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rosgraph_msgs REQUIRED)

ament_target_dependencies(clock_rate rclcpp rosgraph_msgs)
target_link_libraries(clock_rate nlohmann_json::nlohmann_json)
