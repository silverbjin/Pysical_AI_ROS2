cmake_minimum_required(VERSION 3.16)
project(ch13_robot_force_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz_ros2_control REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_library(CH13_OBSERVATION_LIBRARY navigation_gazebo_observation PATHS "${CH13_SENSOR_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
add_executable(robot_force_probe robot_force_probe.cpp)
target_include_directories(robot_force_probe PRIVATE "${CH13_SENSOR_INCLUDE}")
target_link_libraries(robot_force_probe PRIVATE gz-sim8::gz-sim8 gz_ros2_control::gz_ros2_control-system
  hardware_interface::hardware_interface pluginlib::pluginlib rclcpp::rclcpp "${CH13_OBSERVATION_LIBRARY}")
